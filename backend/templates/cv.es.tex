\documentclass[10pt]{article}

\setmainfont{Lato}[

  ItalicFont     = *-Italic,
  BoldFont       = *-Bold,
  BoldItalicFont = *-BoldItalic,
]

\newfontfamily\latoblack{Lato}[UprightFont=*-Black]
\setmonofont{DejaVu Sans Mono}

% --- Sección estilo Harvard: MAYÚSCULAS sobrias + línea fina abajo ---
\newcommand{\cvsection}[1]{%
  \vspace{5pt}%
  {\fontsize{10}{12}\selectfont\bfseries\color{bodytext}\addfontfeature{LetterSpace=6.0}\MakeUppercase{#1}}\par%
  \vspace{3pt}%
  \hrule height 0.4pt%
  \vspace{3pt}%
}

\setlist{itemsep=0pt, topsep=1pt, parsep=0pt, leftmargin=1em}
\pagestyle{empty}
% Interlineado un poco más apretado para ganar líneas extra sin sacrificar lectura.
\linespread{0.98}

\renewcommand{\arraystretch}{1.05}

% Línea de entrada para experiencia/educación: izquierda con título y subtítulo,
% derecha con fecha y ubicación. Argumentos: {izq-bold}{izq-italic}{der-bold}{der-italic}.
\newcommand{\entry}[4]{%
  \begin{tabularx}{\linewidth}{@{}X r@{}}
    \textbf{#1} & \textbf{#3} \\
    \textit{\color{mutedtext}#2} & \textit{\color{mutedtext}#4} \\
  \end{tabularx}
  \vspace{0pt}
}
