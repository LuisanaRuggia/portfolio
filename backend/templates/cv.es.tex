% --- Sección estilo Harvard: MAYÚSCULAS sobrias + línea fina abajo ---
% Jerarquía sutil: 11pt bold con tracking ancho, no `\large` que era muy grande.
\newcommand{\cvsection}[1]{%
  \vspace{10pt}%
  {\fontsize{10.5}{12}\selectfont\bfseries\color{bodytext}\addfontfeature{LetterSpace=6.0}\MakeUppercase{#1}}\par%
  \vspace{1pt}%
  \hrule height 0.4pt%
  \vspace{4pt}%
}

\renewcommand{\arraystretch}{1.15}

% Línea de entrada para experiencia/educación: izquierda con título y subtítulo,
% derecha con fecha y ubicación. Argumentos: {izq-bold}{izq-italic}{der-bold}{der-italic}.
\newcommand{\entry}[4]{%
  \begin{tabularx}{\linewidth}{@{}X r@{}}
    \textbf{#1} & \textbf{#3} \\
    \textit{\color{mutedtext}#2} & \textit{\color{mutedtext}#4} \\
  \end{tabularx}
  \vspace{2pt}
}
